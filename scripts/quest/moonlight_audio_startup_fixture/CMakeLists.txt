cmake_minimum_required(VERSION 3.20)
project(quest3d_moonlight_audio_startup_fixture C CXX)
include(CheckCXXSourceCompiles)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
endif()
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fsanitize=address,undefined -fno-omit-frame-pointer")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -fsanitize=address,undefined")
include_directories(SYSTEM "${OPENSSL_INCLUDE_DIR}" "${HOST_MULTIARCH_INCLUDE}")
add_subdirectory("${MOONLIGHT_SOURCE}" moonlight)
foreach(name audio-startup-test audio-startup-production-denial-test)
  add_executable(${name} test_audio_startup.c)
  target_compile_definitions(${name} PRIVATE HAS_SOCKLEN_T LC_DEBUG)
  target_include_directories(${name} PRIVATE "${MOONLIGHT_SOURCE}/src" "${MOONLIGHT_SOURCE}/enet/include" "${OPUS_INCLUDE_DIR}")
  target_link_libraries(${name} PRIVATE moonlight-common-c enet pthread "${OPENSSL_CRYPTO_LIBRARY}" /lib/x86_64-linux-gnu/libopus.so.0 m)
endforeach()
target_compile_definitions(audio-startup-test PRIVATE QUEST3D_AUDIO_STARTUP_FIXTURE)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_definitions(audio-startup-test PRIVATE QUEST3D_TEST_RELEASE_FEC)
endif()
